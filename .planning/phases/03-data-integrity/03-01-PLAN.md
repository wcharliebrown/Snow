        $snap = [['id' => 1, 'name' => 'Alice', 'age' => '30'], ['id' => 2, 'name' => 'Bob', 'age' => '25']];
        $live = [['id' => 1, 'name' => 'Alice', 'age' => '31'], ['id' => 2, 'name' => 'Bob', 'age' => '25']];
